<script>
let currentOperandText = '0';
let previousOperandText = '';
let operation = undefined;
let resetScreen = false;
let history = [];

const currentOperandElement = document.getElementById('current-operand');
const previousOperandElement = document.getElementById('previous-operand');
const historyListElement = document.getElementById('history-list');

function showToast(message) {
const toast = document.getElementById('toast');
const msg = document.getElementById('toast-message');
msg.textContent = message;
toast.classList.remove('opacity-0', 'pointer-events-none');
toast.classList.add('opacity-100');
setTimeout(() => {
toast.classList.remove('opacity-100');
toast.classList.add('opacity-0', 'pointer-events-none');
}, 3000);
}

function appendNumber(number) {
if (currentOperandText === '0' || resetScreen) {
currentOperandText = '';
resetScreen = false;
}
if (number === '.' && currentOperandText.includes('.')) return;
currentOperandText += number;
updateDisplay();
}

function appendOperation(op) {
if (currentOperandText === '') return;
if (previousOperandText !== '') {
compute();
}
operation = op;
previousOperandText = currentOperandText + ' ' + getSymbol(op);
currentOperandText = '';
updateDisplay();
}

function getSymbol(op) {
switch(op) {
case '+': return '+';
case '-': return '-';
case '*': return '×';
case '/': return '÷';
case '%': return '%';
default: return '';
}
}

function compute() {
let computation;
const prev = parseFloat(previousOperandText);
const current = parseFloat(currentOperandText);

if (isNaN(prev) || isNaN(current)) return;
const op = previousOperandText.slice(-1);

switch (op) {
case '+': computation = prev + current; break;
case '-': computation = prev - current; break;
case '×': computation = prev * current; break;
case '÷':
if (current === 0) {
showToast('لا يمكن القسمة على الصفر!');
clearAll();
return;
}
computation = prev / current;
break;
default: return;
}

computation = Math.round(computation * 1000000) / 1000000;
const calculationString = `${previousOperandText} ${currentOperandText} = ${computation}`;
addToHistory(calculationString);

currentOperandText = computation.toString();
operation = undefined;
previousOperandText = '';
resetScreen = true;
updateDisplay();
}

function clearAll() {
currentOperandText = '0';
previousOperandText = '';
operation = undefined;
updateDisplay();
}

function deleteDigit() {
if (resetScreen) return;
currentOperandText = currentOperandText.toString().slice(0, -1);
if (currentOperandText === '' || currentOperandText === '-') {
currentOperandText = '0';
}
updateDisplay();
}

function updateDisplay() {
currentOperandElement.textContent = currentOperandText;
previousOperandElement.textContent = previousOperandText;
}

function addToHistory(item) {
history.unshift(item);
renderHistory();
}

function renderHistory() {
if (history.length === 0) {
historyListElement.innerHTML = `
<div class="h-full flex flex-col items-center justify-center text-slate-500 text-sm text-center">
<i class="fa-solid fa-receipt text-3xl mb-2 opacity-30"></i>
<span>لا توجد عمليات سابقة حتى الآن</span>
</div>`;
return;
}

historyListElement.innerHTML = '';
history.forEach((calc) => {
const parts = calc.split('=');
const expression = parts[0];
const result = parts[1];

const div = document.createElement('div');
div.className = 'bg-white/5 border border-white/5 rounded-xl p-3 text-sm hover:bg-white/10 transition-all cursor-pointer';
div.innerHTML = `
<div class="text-slate-400 text-xs">${expression}</div>
<div class="text-white font-bold text-base mt-0.5">= ${result}</div>
`;
div.onclick = () => {
currentOperandText = result.trim();
resetScreen = true;
updateDisplay();
};
historyListElement.appendChild(div);
});
}

function clearHistory() {
history = [];
renderHistory();
}

function toggleHistoryMobile() {
const panel = document.getElementById('history-panel');
panel.classList.toggle('hidden');
}
